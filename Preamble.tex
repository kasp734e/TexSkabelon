% --- Language, fonts and typography ---
\usepackage{fontspec}                      % system fonts (requires LuaLaTeX, see .latexmkrc)
% AU Passata (installed in ~/Library/Fonts) as the main font. It has no italic,
% so \emph/\textit use a slanted version of the upright font.
% Falls back to the default font (Latin Modern) if AU Passata isn't installed.
\IfFontExistsTF{AU Passata}{
\setmainfont{AU Passata}[
    ItalicFont         = AU Passata,
    ItalicFeatures     = {FakeSlant = 0.2},
    BoldItalicFont     = AU Passata Bold,
    BoldItalicFeatures = {FakeSlant = 0.2}
]}{}
